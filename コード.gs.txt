function doGet() {
  var output = HtmlService.createHtmlOutputFromFile('index');
  output.addMetaTag('viewport', 'width=device-width, initial-scale=1');
  output.setTitle('契約管理システム');
  return output;
}

// ▼▼▼ 1. ログイン判定（勝手に登録しない版） ▼▼▼
function getLoginUser() {
  var email = Session.getActiveUser().getEmail();
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("ユーザー管理");
  var data = sheet.getDataRange().getValues();
  
  var userName = "";
  var isRegistered = false;
  
  // 名簿にあるかチェック
  for (var i = 1; i < data.length; i++) {
    if (data[i][0] === email) {
      userName = data[i][2];
      isRegistered = true;
      break;
    }
  }
  
  // 結果を返す（未登録なら isRegistered は false のまま）
  return {
    email: email,
    name: userName,
    isRegistered: isRegistered
  };
}

// ▼▼▼ 2. ★追加：名前を指定してユーザー登録する関数 ▼▼▼
function registerGoogleUser(inputName) {
  var email = Session.getActiveUser().getEmail(); // メアドは自動取得（偽装防止）
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("ユーザー管理");
  
  // 重複チェック（念のため）
  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (data[i][0] === email) {
      return "ALREADY_REGISTERED";
    }
  }
  
  // 保存！ A列:メアド, B列:ダミー, C列:入力された名前
  sheet.appendRow([email, "google_login", inputName]);
  
  return "SUCCESS";
}

// ▼▼▼ 契約リスト取得（既存のまま） ▼▼▼
function getContractList(targetEmail) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("契約リスト");
  var data = sheet.getDataRange().getDisplayValues(); 
  var myContracts = [];

  for (var i = 1; i < data.length; i++) {
    if (data[i][0] === targetEmail) {
      myContracts.push({
        row: i + 1,
        project:    data[i][1],
        contractor: data[i][2],
        referrer:   data[i][3],
        approval:   data[i][4],
        expiryDate: data[i][5],
        application: data[i][6],
        amount:     data[i][7],
        memo:       data[i][8]
      });
    }
  }
  return myContracts;
}

// ▼▼▼ 更新機能（既存のまま） ▼▼▼
function updateContract(row, project, contractor, referrer, approval, expiryDate, application, amount, memo) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("契約リスト");
  sheet.getRange(row, 2, 1, 8).setValues([[
    project, contractor, referrer, approval, expiryDate, application, amount, memo
  ]]);
  return "SUCCESS";
}
