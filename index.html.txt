    <script>
      // 画面切り替え：登録へ
      function showRegister() {
        document.getElementById("login-screen").style.display = "none";
        document.getElementById("register-screen").style.display = "flex";
        document.getElementById("msg-login").innerHTML = "";
      }

      // 画面切り替え：ログインへ
      function showLogin() {
        document.getElementById("register-screen").style.display = "none";
        document.getElementById("login-screen").style.display = "flex";
        document.getElementById("msg-reg").innerHTML = "";
      }

      // ログイン処理
      function startLogin() {
        var email = document.getElementById("email").value;
        var pass = document.getElementById("password").value;
        var msg = document.getElementById("msg-login");
        msg.innerHTML = '<div class="loader"></div>';

        google.script.run
          .withSuccessHandler(function(resultName) {
            if (resultName) {
              document.getElementById("login-screen").style.display = "none";
              document.getElementById("main-screen").style.display = "flex";
              document.getElementById("user-name").innerText = resultName;
              msg.innerHTML = "";
            } else {
              msg.innerText = "😱 エラー：メールかパスワードが違います";
              msg.style.color = "red";
            }
          })
          .checkLogin(email, pass);
      }

      // ★追加：登録処理
      function startRegister() {
        var name = document.getElementById("reg-name").value;
        var email = document.getElementById("reg-email").value;
        var pass = document.getElementById("reg-password").value;
        var msg = document.getElementById("msg-reg");

        if (!name || !email || !pass) {
          msg.innerText = "⚠️ 全部の項目を入力してください";
          msg.style.color = "orange";
          return;
        }

        msg.innerHTML = '<div class="loader"></div>';

        google.script.run
          .withSuccessHandler(function(result) {
            if (result === "SUCCESS") {
              // 成功したらそのままログイン状態にする！
              document.getElementById("register-screen").style.display = "none";
              document.getElementById("main-screen").style.display = "flex";
              document.getElementById("user-name").innerText = name;
              msg.innerHTML = "";
            } else if (result === "ERROR_EXIST") {
              msg.innerText = "😱 そのメールアドレスは既に登録されています";
              msg.style.color = "red";
            } else {
              msg.innerText = "😱 エラーが発生しました";
              msg.style.color = "red";
            }
          })
          .registerUser(email, pass, name);
      }

      function logout() {
        location.reload();
      }

      // 共通：パスワード表示切り替え（IDを指定できるようにパワーアップ！）
      function togglePass(inputId, iconOpenId, iconClosedId) {
        var input = document.getElementById(inputId);
        var iconOpen = document.getElementById(iconOpenId);
        var iconClosed = document.getElementById(iconClosedId);

        if (input.type === "password") {
          input.type = "text";
          iconOpen.style.display = "none";
          iconClosed.style.display = "block";
        } else {
          input.type = "password";
          iconOpen.style.display = "block";
          iconClosed.style.display = "none";
        }
      }
    </script>
