    <script>
      var currentUserEmail = "";
      var allContracts = [];

      // ★便利機能：きれいなローダーのHTMLを返す関数
      function getLoaderHtml() {
        return '<svg class="loader" viewBox="0 0 50 50"><circle class="path" cx="25" cy="25" r="20" fill="none" stroke-width="5"></circle></svg>';
      }

      window.onload = function() {
        google.script.run.withSuccessHandler(function(user) {
          document.getElementById("loading-screen").style.display = "none";
          currentUserEmail = user.email;

          if (user.isRegistered) {
            showMainScreen(user.name);
          } else {
            document.getElementById("signup-screen").style.display = "flex";
            document.getElementById("signup-email").innerText = user.email;
          }
        }).getLoginUser();
      };

      function doSignup() {
        var name = document.getElementById("signup-name").value;
        if (!name) return;

        // ★ここでSVGローダーを使う
        document.getElementById("signup-screen").innerHTML = getLoaderHtml() + '<p style="text-align:center; margin-top:20px;">登録中...</p>';
        
        google.script.run
          .withSuccessHandler(function(res) {
             document.getElementById("signup-screen").style.display = "none";
             showMainScreen(name);
          })
          .registerGoogleUser(name);
      }

      function showMainScreen(name) {
        document.getElementById("main-screen").style.display = "flex";
        document.getElementById("user-name").innerText = name;
        loadContracts();
      }

      function loadContracts() {
        var listContainer = document.getElementById("contract-list");
        // ★ここでもSVGローダーを使う
        listContainer.innerHTML = getLoaderHtml();

        google.script.run.withSuccessHandler(function(data) {
           allContracts = data;
           listContainer.innerHTML = "";
           if (data.length === 0) {
             listContainer.innerHTML = '<p class="no-data">まだ契約データがありません</p>';
           } else {
             for (var i = 0; i < data.length; i++) {
               var item = data[i];
               var html = `
                  <div class="contract-item">
                    <div class="edit-btn" onclick="openEdit(${i})" title="編集">
                      <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" width="20" height="20">
                        <path d="M3 17.25V21h3.75L17.81 9.94l-3.75-3.75L3 17.25zM20.71 7.04c.39-.39.39-1.02 0-1.41l-2.34-2.34c-.39-.39-1.02-.39-1.41 0l-1.83 1.83 3.75 3.75 1.83-1.83z"/><path d="M0 0h24v24H0z" fill="none"/>
                      </svg>
                    </div>
                    <span class="project-name">${item.project}</span>
                    <div class="info-row"><span class="info-label">契約者:</span><span class="info-text">${item.contractor}</span><span style="margin:0 8px; color:#ddd;">|</span><span class="info-label">紹介元:</span><span class="info-text">${item.referrer}</span></div>
                    <div class="info-row"><span class="info-label">金額:</span><span class="info-text" style="font-weight:bold;">${item.amount}</span><span style="margin:0 8px; color:#ddd;">|</span><span class="info-label">満了日:</span><span class="info-text">${item.expiryDate}</span></div>
                    <div class="info-row"><span class="info-label">メモ:</span><span class="info-text">${item.memo}</span></div>
                    <div class="tags-row"><span class="tag">稟議: ${item.approval}</span><span class="tag">申請: ${item.application}</span></div>
                  </div>`;
               listContainer.insertAdjacentHTML('beforeend', html);
             }
           }
        }).getContractList(currentUserEmail);
      }

      function openEdit(index) {
        var item = allContracts[index];
        document.getElementById("edit-row").value = item.row;
        document.getElementById("edit-project").value = item.project;
        document.getElementById("edit-contractor").value = item.contractor;
        document.getElementById("edit-referrer").value = item.referrer;
        document.getElementById("edit-approval").value = item.approval;
        document.getElementById("edit-application").value = item.application;
        document.getElementById("edit-expiry").value = item.expiryDate;
        document.getElementById("edit-amount").value = item.amount;
        document.getElementById("edit-memo").value = item.memo;
        document.getElementById("edit-modal").style.display = "flex";
      }

      function saveEdit() {
        var row = document.getElementById("edit-row").value;
        var project = document.getElementById("edit-project").value;
        var contractor = document.getElementById("edit-contractor").value;
        var referrer = document.getElementById("edit-referrer").value;
        var approval = document.getElementById("edit-approval").value;
        var application = document.getElementById("edit-application").value;
        var expiryDate = document.getElementById("edit-expiry").value;
        var amount = document.getElementById("edit-amount").value;
        var memo = document.getElementById("edit-memo").value;
        var msg = document.getElementById("msg-edit");
        
        // ★ここでもSVGローダーを使う
        msg.innerHTML = getLoaderHtml();

        google.script.run.withSuccessHandler(function(res) {
          if (res === "SUCCESS") { closeEdit(); loadContracts(); }
          else { msg.innerText = "エラーが発生しました"; }
        }).updateContract(row, project, contractor, referrer, approval, expiryDate, application, amount, memo);
      }

      function closeEdit() {
        document.getElementById("edit-modal").style.display = "none";
        document.getElementById("msg-edit").innerText = "";
      }
    </script>
